def load_exports():
    exports = []
    canonical_lilos = {}
    seen_questions = set()

    for path in sorted(JSONS.glob("*.json")):
        data = json.loads(path.read_text(encoding="utf-8"))
        questions = data.get("questions")
        pool = data.get("enrichedLlos")
        if not isinstance(questions, list) or not isinstance(pool, list):
            print(f"Skipping {path.name}: no questions/enrichedLlos arrays")
            continue

        catalog = []
        for item in pool:
            lilo_id = str(item.get("id") or "").strip()
            text = str(item.get("text") or "").strip()
            if not lilo_id or not text:
                raise ValueError(f"{path.name} has a LILO without id/text")
            if lilo_id in canonical_lilos and canonical_lilos[lilo_id] != text:
                raise ValueError(f"Conflicting text for {lilo_id} in {path.name}")
            canonical_lilos[lilo_id] = text
            catalog.append({"lilo_id": lilo_id, "lilo": text})

        pool_ids = {item["lilo_id"] for item in catalog}
        prepared = []
        for question in questions:
            text = str(question.get("question") or "").strip()
            gold = str(question.get("source_ilo_id") or "").strip()
            if not text or not gold:
                raise ValueError(f"{path.name} has a question without question/source_ilo_id")
            if gold not in pool_ids:
                raise ValueError(f"{path.name}: source label {gold} is absent from its pool")
            normalized = " ".join(text.lower().split())
            if normalized in seen_questions:
                raise ValueError(f"Duplicate question text found in {path.name}")
            seen_questions.add(normalized)
            prepared.append({
                "question_id": f"{path.stem}-{question.get('index', len(prepared) + 1)}",
                "question": text,
                "gold_lilo_id": gold,
                "question_type": str(question.get("type") or ""),
                "export": path.name,
                "saved_alignment_lilo_id": str((question.get("alignment") or {}).get("lilo_id") or ""),
            })

        stated_total = (data.get("stats") or {}).get("total")
        if stated_total is not None and stated_total != len(prepared):
            raise ValueError(f"{path.name}: stats.total does not match questions")
        exports.append({"name": path.name, "questions": prepared, "catalog": catalog})

    if not exports:
        raise ValueError(f"No compatible assessment exports found in {JSONS}")
    union_catalog = [{"lilo_id": key, "lilo": value} for key, value in sorted(
        canonical_lilos.items(), key=lambda item: int(item[0].split("-")[-1])
    )]
    return exports, union_catalog


def score_pairs(model, tokenizer, pairs):
    scores = []
    with torch.inference_mode():
        for start in range(0, len(pairs), BATCH_SIZE):
            batch = pairs[start:start + BATCH_SIZE]
            encoded = tokenizer(
                [item[0] for item in batch],
                [item[1] for item in batch],
                padding=True,
                truncation=True,
                max_length=MAX_LENGTH,
                return_tensors="pt",
            )
            logits = model(**{key: value.to(model.device) for key, value in encoded.items()}).logits
            values = logits[:, 0] if logits.ndim == 2 and logits.shape[1] == 1 else logits[:, -1]
            scores.extend(values.detach().cpu().tolist())
    return scores


def rank_questions(model, tokenizer, questions, catalog, condition):
    catalog_by_id = {item["lilo_id"]: item for item in catalog}
    flat_pairs = [(question["question"], candidate["lilo"]) for question in questions for candidate in catalog]
    flat_scores = score_pairs(model, tokenizer, flat_pairs)
    width = len(catalog)
    details = []

    for index, question in enumerate(questions):
        gold = question["gold_lilo_id"]
        if gold not in catalog_by_id:
            raise ValueError(f"{question['question_id']}: {gold} is not in the candidate catalog")
        scores = flat_scores[index * width:(index + 1) * width]
        ranked = sorted(zip(scores, catalog), key=lambda item: item[0], reverse=True)
        rank = next(position for position, (_, item) in enumerate(ranked, 1) if item["lilo_id"] == gold)
        top = ranked[:5]
        details.append({
            **question,
            "condition": condition,
            "candidate_pool_size": width,
            "gold_lilo_text": catalog_by_id[gold]["lilo"],
            "positive_rank": rank,
            "predicted_lilo_id": top[0][1]["lilo_id"],
            "predicted_lilo_text": top[0][1]["lilo"],
            "top5_lilo_ids": "|".join(item[1]["lilo_id"] for item in top),
            "top5_lilo_texts": " | ".join(item[1]["lilo"] for item in top),
            "top5_scores": "|".join(f"{item[0]:.6f}" for item in top),
            "hit_at_1": int(rank == 1),
            "hit_at_3": int(rank <= 3),
            "hit_at_5": int(rank <= 5),
        })
    return details


def metrics(rows):
    count = len(rows)
    return {
        "questions": count,
        "top1_hits": sum(row["hit_at_1"] for row in rows),
        "top1_agreement": round(sum(row["hit_at_1"] for row in rows) / count, 4),
        "mrr": round(sum(1 / row["positive_rank"] for row in rows) / count, 4),
        "recall_at_3": round(sum(row["hit_at_3"] for row in rows) / count, 4),
        "recall_at_5": round(sum(row["hit_at_5"] for row in rows) / count, 4),
    }


def per_lilo_metrics(rows, catalog):
    ids = [item["lilo_id"] for item in catalog]
    text = {item["lilo_id"]: item["lilo"] for item in catalog}
    true_counts = Counter(row["gold_lilo_id"] for row in rows)
    pred_counts = Counter(row["predicted_lilo_id"] for row in rows)
    true_positive = Counter(
        row["gold_lilo_id"] for row in rows if row["gold_lilo_id"] == row["predicted_lilo_id"]
    )
    result = []
    for lilo_id in ids:
        tp = true_positive[lilo_id]
        support = true_counts[lilo_id]
        predicted = pred_counts[lilo_id]
        precision = tp / predicted if predicted else 0.0
        recall = tp / support if support else 0.0
        f1 = 2 * precision * recall / (precision + recall) if precision + recall else 0.0
        result.append({
            "lilo_id": lilo_id,
            "lilo_text": text[lilo_id],
            "support": support,
            "predicted": predicted,
            "true_positive": tp,
            "precision": round(precision, 4),
            "recall": round(recall, 4),
            "f1": round(f1, 4),
        })
    return result


def save_csv(path, rows):
    if not rows:
        return
    with path.open("w", encoding="utf-8", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)
